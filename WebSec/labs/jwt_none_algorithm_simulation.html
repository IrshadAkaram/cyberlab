🛡️ How To Defend: Enforcing Explicit Algorithm Whitelisting

Server ko kabhi bhi aane wale token ke header par andha bharosa nahi karna chahiye. Hamesha verification function mein explicitly algorithms: ['HS256'] define karein taaki library none ko turant reject kar de.

❌ VULNERABLE CODE (Accepts whatever alg header claims)

// Insecure: Library defaults allow "none"
const jwt = require('jsonwebtoken');

function authenticate(req, res, next) {
  const token = req.headers.authorization?.split(' ')[1];
  
  // FLAW: No algorithms array specified!
  // If header has "alg": "none", verification succeeds!
  jwt.verify(token, SERVER_SECRET, (err, user) => {
    if (err) return res.sendStatus(403);
    req.user = user;
    next();
  });
}

✅ SECURE REMEDIATION (Strict Algorithm Pinning)

// Secure: Explicit algorithm pinning
const jwt = require('jsonwebtoken');

function authenticate(req, res, next) {
  const token = req.headers.authorization?.split(' ')[1];
  
  // FIX: Force algorithm whitelist and reject "none"
  jwt.verify(token, SERVER_SECRET, {
    algorithms: ['HS256'], // Explicitly whitelist allowed algs
    maxAge: '1h'
  }, (err, user) => {
    if (err) return res.status(403).json({ error: "Invalid signature" });
    req.user = user;
    next();
  });
}
