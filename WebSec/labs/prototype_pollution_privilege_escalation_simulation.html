🛡️ How To Defend: Sanitizing Prototype Keys & Freezing Prototypes

Recursive merge functions mein hamesha check karein ki key __proto__, constructor, ya prototype toh nahi hai. Saath hi startup par Object.freeze(Object.prototype) laga kar prototype ko immutable banayein.

❌ VULNERABLE CODE (Naive Recursive Merge)

// Vulnerable recursive merge
function badMerge(target, source) {
  for (let key in source) {
    if (typeof source[key] === 'object') {
      if (!target[key]) target[key] = {};
      badMerge(target[key], source[key]); // FLAW!
    } else {
      // If key is __proto__, mutates Object.prototype!
      target[key] = source[key];
    }
  }
  return target;
}

✅ SECURE REMEDIATION (Key Filtering & Freeze)

// Secure deep merge
function safeMerge(target, source) {
  for (let key in source) {
    // FIX 1: Filter dangerous prototype traversal keys
    if (key === '__proto__' || key === 'constructor' || key === 'prototype') {
      continue;
    }
    if (typeof source[key] === 'object' && source[key] !== null) {
      if (!target[key]) target[key] = {};
      safeMerge(target[key], source[key]);
    } else {
      target[key] = source[key];
    }
  }
  return target;
}

// FIX 2: Immutable prototype
Object.freeze(Object.prototype);
