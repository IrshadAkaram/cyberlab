# 2. Application Layer: Strict Protocol & Link-Local CIDR Validation
def safe_fetch_url(url):
    parsed = urllib.parse.urlparse(url)
    if parsed.scheme not in ('http', 'https'):
        raise ValueError("Invalid URL scheme")

    ip = socket.gethostbyname(parsed.hostname)
    ip_obj = ipaddress.ip_address(ip)

    # ✅ Block Private IPs & Link-Local Cloud Metadata (169.254.0.0/16)
    if ip_obj.is_private or ip_obj.is_link_local or ip_obj.is_loopback:
        raise PermissionError("Access to internal/metadata IPs prohibited")

    return requests.get(url, timeout=3)
