🛡️ How To Defend: Row-Level Locking (SELECT ... FOR UPDATE)

Non-atomic check-then-act logic race conditions ko janam deta hai. Transaction ke andar SELECT ... FOR UPDATE use karein taaki jab tak pehla thread deduction poora karke commit na kare, tab tak doosra thread row ko chhoo bhi na sake.

❌ VULNERABLE CODE (Non-Atomic Check & Update)

// Insecure: Race window exists between SELECT & UPDATE
async function withdraw(userId, amount) {
  // Step 1: Check
  const user = await db.query(
    "SELECT balance FROM users WHERE id = $1", [userId]
  );
  
  // ⏱️ RACE WINDOW: Concurrent thread reads $100 here!
  if (user.balance >= amount) {
    // Step 2: Act
    await payoutProvider.send(amount);
    await db.query(
      "UPDATE users SET balance = balance - $1 WHERE id = $2",
      [amount, userId]
    );
  }
}

✅ SECURE REMEDIATION (Pessimistic Row Lock)

// Secure: Transaction with SELECT ... FOR UPDATE
async function withdraw(userId, amount) {
  await db.query("BEGIN TRANSACTION");
  
  // FIX: Locks the row exclusively until COMMIT!
  const user = await db.query(
    "SELECT balance FROM users WHERE id = $1 FOR UPDATE",
    [userId]
  );
  
  if (user.balance >= amount) {
    await payoutProvider.send(amount);
    await db.query(
      "UPDATE users SET balance = balance - $1 WHERE id = $2",
      [amount, userId]
    );
    await db.query("COMMIT");
  } else {
    await db.query("ROLLBACK");
    throw new Error("Insufficient Funds");
  }
}
