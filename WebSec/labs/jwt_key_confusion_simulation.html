🛡️ How To Defend: Never Mix Asymmetric & Symmetric Key Handlers

Server ko public key se verify karte waqt hamesha algorithms: ['RS256'] pass karna chahiye taaki agar token header mein HS256 ho, toh library turant algorithm mismatch error phenk de.

❌ VULNERABLE CODE (Algorithm Confusion)

// Insecure: Uses publicKey as secret for HS256!
const fs = require('fs');
const jwt = require('jsonwebtoken');
const publicKey = fs.readFileSync('public.pem');

app.post('/api/profile', (req, res) => {
  const token = req.headers.authorization.split(' ')[1];
  
  // FLAW: If token says "alg": "HS256", jwt.verify
  // treats publicKey string as the HMAC secret!
  jwt.verify(token, publicKey, (err, decoded) => {
    if (err) return res.sendStatus(401);
    res.json(decoded);
  });
});

✅ SECURE REMEDIATION (Enforce RS256)

// Secure: Algorithm pinned to RS256 only
const fs = require('fs');
const jwt = require('jsonwebtoken');
const publicKey = fs.readFileSync('public.pem');

app.post('/api/profile', (req, res) => {
  const token = req.headers.authorization.split(' ')[1];
  
  // FIX: Force library to only accept RS256
  jwt.verify(token, publicKey, {
    algorithms: ['RS256'] // HS256 will be REJECTED!
  }, (err, decoded) => {
    if (err) return res.status(401).json({ error: "Invalid Algorithm" });
    res.json(decoded);
  });
});
